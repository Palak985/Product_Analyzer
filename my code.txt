def clean_data(text):
    if not isinstance(text, str):
        return ""
    text = re.sub(r'[^a-zA-Z0-9\s.,!?\'"]', '', text)
    return " ".join(text.split()).strip()

def preprocess_text(text):
    if not text or pd.isna(text):
        return ""
    text = text.lower()
    text = re.sub(r'[^a-z\s]', '', text)
    try:
        words = [lemmatizer.lemmatize(w) for w in text.split() if w not in stop_words]
    except Exception:
        words = [w for w in text.split() if w not in stop_words]
    return " ".join(words).strip()

def ml_sentiment(text):
    score = vader.polarity_scores(text)['compound']
    if score >= 0.05:
        return 'Positive'
    elif score <= -0.05:
        return 'Negative'
    else:
        return 'Neutral'

def analyze_sentiment(df):
    neg_words = {'bad','worst','poor','terrible','horrible','waste','hate','disappointed','cheap'}
    pos_words = {'good','great','excellent','amazing','love','perfect','nice','beautiful','best'}

    sentiments = []
    sentiment_scores = []

    for text in df['preprocessed_review']:
        words = set(text.split())
        if words & neg_words:
            sentiments.append('Negative')
            sentiment_scores.append(-1)
        elif words & pos_words:
            sentiments.append('Positive')
            sentiment_scores.append(1)
        else:
            sentiments.append('Neutral')
            sentiment_scores.append(0)

    df['sentiment'] = sentiments
    df['sentiment_score'] = sentiment_scores

    counts = df['sentiment'].value_counts().to_dict()
    total = len(df)
    pos = counts.get('Positive', 0)
    neg = counts.get('Negative', 0)
    neu = counts.get('Neutral', 0)
    pos_percent = round((pos / total) * 100, 2) if total else 0

    if neg / total > 0.25:
        recommendation = "Not Recommended"
    elif pos_percent > 75:
        recommendation = "Highly Recommended"
    else:
        recommendation = "Needs Improvement"

    # Create the results dictionary **first**
    results = {
        "total_reviews": total,
        "sentiment_counts": {"Positive": pos, "Negative": neg, "Neutral": neu},
        "positive_percent": pos_percent,
        "recommendation": recommendation
    }

    # ---------------- TREND DATA ----------------
    # Use index as x-axis
    df['index'] = range(1, len(df) + 1)
    trend_positive = df['sentiment'].eq('Positive').astype(int).cumsum().tolist()
    trend_negative = df['sentiment'].eq('Negative').astype(int).cumsum().tolist()
    trend_neutral = df['sentiment'].eq('Neutral').astype(int).cumsum().tolist()
    trend_dates = df['index'].tolist()

    # Add trend data to results
    results['trend_dates'] = trend_dates
    results['trend_positive'] = trend_positive
    results['trend_negative'] = trend_negative
    results['trend_neutral'] = trend_neutral

    return results, df

@app.route('/')

@app.route('/register', methods=['GET','POST'])
def register():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        if User.query.filter_by(username=username).first():
            flash("Username already exists")
            return redirect(url_for('register'))
        user = User(username=username, password=generate_password_hash(password))
        db.session.add(user)
        db.session.commit()
        return redirect(url_for('login'))
    return render_template('register.html')

@app.route('/login', methods=['GET','POST'])
def login():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        user = User.query.filter_by(username=username).first()
        if user and check_password_hash(user.password, password):
            login_user(user)
            return redirect(url_for('dashboard'))
        flash("Invalid login")
    return render_template('login.html')

@app.route('/upload', methods=['POST'])
@login_required
def upload_file():
    file = request.files.get('csv_file')
    if not file or file.filename == "":
        flash("No file selected")
        return redirect(url_for('dashboard'))

    try:
        df = pd.read_csv(file, encoding='utf-8', on_bad_lines='skip')

        return redirect(url_for('dashboard'))

    df.columns = df.columns.str.strip()
    possible_cols = ['Review_Text', 'Review Text', 'review_body', 'text', 'Review', 'Body', 'Comments']
    col = next((c for c in possible_cols if c in df.columns), None)

    if not col:
        flash("Review column not found. Use 'Review_Text'.")
        return redirect(url_for('dashboard'))

    df = df.rename(columns={col: 'Review_Text'})
    df = df.dropna(subset=['Review_Text'])
    df['Review_Text'] = df['Review_Text'].apply(clean_data)
    df = df[df['Review_Text'] != ""]
    df['preprocessed_review'] = df['Review_Text'].apply(preprocess_text)

    results, df = analyze_sentiment(df)
    df['ml_sentiment'] = df['Review_Text'].apply(ml_sentiment)

    # Save analysis history in DB
    overall_sentiment = max(results['sentiment_counts'], key=results['sentiment_counts'].get)
    history_entry = AnalysisHistory(user_id=current_user.id, filename=file.filename, overall_sentiment=overall_sentiment)
    db.session.add(history_entry)
    db.session.commit()

    # Save CSV
    output_path = os.path.join(app.config['UPLOAD_FOLDER'], 'analysis_report.csv')
    df.to_csv(output_path, index=False)

    return render_template('analysis_report.html', results=results, review_data=df.to_dict('records'), filename=file.filename)

@app.route('/view_result/<int:id>')
@login_required
def view_result(id):
    # Get the analysis history item or 404 if not found
    history_item = AnalysisHistory.query.get_or_404(id)

    # Ensure only the owner can view
    if history_item.user_id != current_user.id:
        flash("Access Denied")
        return redirect(url_for('dashboard'))

    # Load the CSV report (if it exists)
    csv_path = os.path.join(app.config['UPLOAD_FOLDER'], 'analysis_report.csv')
    if os.path.exists(csv_path):
        df = pd.read_csv(csv_path)
        review_data = df.to_dict('records')

        # Recalculate results for the template
        counts = df['sentiment'].value_counts().to_dict()
        total = len(df)
        pos = counts.get('Positive', 0)
        neg = counts.get('Negative', 0)
        neu = counts.get('Neutral', 0)
        pos_percent = round((pos / total) * 100, 2) if total else 0

        if neg / total > 0.25:
            recommendation = "Not Recommended"
        elif pos_percent > 75:
            recommendation = "Highly Recommended"
        else:
            recommendation = "Needs Improvement"

        results = {
            "total_reviews": total,
            "sentiment_counts": {"Positive": pos, "Negative": neg, "Neutral": neu},
            "positive_percent": pos_percent,
            "recommendation": recommendation,
            "dominant_sentiment": max(counts, key=counts.get)
        }
    else:
        review_data = []
        results = {
            "total_reviews": 0,
            "sentiment_counts": {"Positive": 0, "Negative": 0, "Neutral": 0},
            "positive_percent": 0,
            "recommendation": "No Data",
            "dominant_sentiment": "Neutral"
        }

    return render_template(
        'analysis_report.html',
        results=results,
        review_data=review_data,
        filename=history_item.filename
    )
@app.route('/history')
@login_required
def history():
    # Get analysis history for the current user
    history_items = AnalysisHistory.query.filter_by(user_id=current_user.id).order_by(AnalysisHistory.created_at.desc()).all()
    
    # Prepare extra fields for template
    history_list = []
    for item in history_items:
        csv_path = os.path.join(app.config['UPLOAD_FOLDER'], 'analysis_report.csv')
        if os.path.exists(csv_path):
            df = pd.read_csv(csv_path)
            counts = df['sentiment'].value_counts().to_dict()
            total = len(df)
            pos = counts.get('Positive', 0)
            pos_percent = round((pos / total) * 100, 2) if total else 0
            if pos_percent > 75:
                recommendation = "Highly Recommended"
            elif pos_percent < 25:
                recommendation = "Not Recommended"
            else:
                recommendation = "Needs Improvement"
        else:
            total = 0
            pos_percent = 0
            recommendation = "No Data"

        history_list.append({
            "filename": item.filename,
            "time": item.created_at.strftime('%d %b %Y, %I:%M %p'),
            "total_reviews": total,
            "positive_percent": pos_percent,
            "recommendation": recommendation
        })

    return render_template('history.html', history=history_list)
