def preprocess_text(text):
    if not text: return ""
    text = str(text).lower()
    # Keep letters and spaces to preserve meaning
    text = re.sub(r'[^a-zA-Z\s]', '', text)
    return text

def analyze_sentiment(df):
    # Expanded triggers to catch specific negative phrases like "waste of money"
    neg_triggers = {'not', 'no', 'bad', 'worst', 'terrible', 'horrible', 'poor', 'disappointed', 'waste', 'money', 'hate', 'itchy', 'cheap'}
    pos_triggers = {'good', 'great', 'best', 'excellent', 'amazing', 'love', 'perfect', 'nice', 'beautiful'}

    results_list = [] 
    for text in df['preprocessed_review']:
        words = set(str(text).lower().split())
        # Priority Logic: Check for Negative triggers first to handle negations correctly
        if any(w in neg_triggers for w in words):
            results_list.append('Negative')
        elif any(w in pos_triggers for w in words):
            results_list.append('Positive')
        else:
            results_list.append('Neutral')

    df['sentiment'] = results_list
    counts = df['sentiment'].value_counts().to_dict()
    total = len(df)
    
    analysis_results = {
        'total_reviews': total,
        'sentiment_counts': {
            'Positive': counts.get('Positive', 0),
            'Negative': counts.get('Negative', 0),
            'Neutral': counts.get('Neutral', 0)
        },
        'positive_percent': round((counts.get('Positive', 0)/total)*100, 2) if total > 0 else 0,
        'recommendation': "Improvement Needed" if counts.get('Negative', 0) > 0 else "Customer Favorite"
    }
    # SMART RECOMMENDATIONS LOGIC
    pos_perc = (counts.get('Positive', 0) / total) * 100 if total > 0 else 0
    neg_perc = (counts.get('Negative', 0) / total) * 100 if total > 0 else 0

    if neg_perc > 25:
        advice = "⚠️ CRITICAL: High customer dissatisfaction. Investigate product quality or shipping issues immediately."
        color = "#e74c3c" # Red
    elif pos_perc > 80:
        advice = "⭐ SUCCESS: Outstanding feedback! This product is a candidate for featured marketing campaigns."
        color = "#2ecc71" # Green
    elif counts.get('Neutral', 0) > (total * 0.5):
        advice = "ℹ️ NOTICE: High neutral volume. Customers find the product 'okay' but not exciting. Consider adding new features."
        color = "#f1c40f" # Yellow
    else:
        advice = "✅ STABLE: Product performance is consistent with market standards. Maintain current operations."
        color = "#3498db" # Blue

    analysis_results['smart_advice'] = advice
    analysis_results['advice_color'] = color
    return analysis_results, df

def clean_data(text):
    """Removes unwanted characters and extra spaces from raw review text."""
    if not isinstance(text, str):
        return ""
    text = re.sub(r'[^a-zA-Z0-9\s.,!?\'"]', '', text)
    return " ".join(text.split()).strip()

def preprocess_text(text):
    """Lowercase, remove non-letter characters, remove stopwords, lemmatize safely."""
    if not text or pd.isna(text):
        return ""
    text = text.lower()
    text = re.sub(r'[^a-z\s]', '', text)
    try:
        words = [lemmatizer.lemmatize(w) for w in text.split() if w not in stop_words]
    except Exception:
        words = [w for w in text.split() if w not in stop_words]
    return " ".join(words).strip()

# -------------------- SENTIMENT ANALYSIS --------------------
vader = SentimentIntensityAnalyzer()

def ml_sentiment(text):
    """Vader sentiment classification."""
    score = vader.polarity_scores(text)['compound']
    if score >= 0.05:
        return 'Positive'
    elif score <= -0.05:
        return 'Negative'
    else:
        return 'Neutral'

def analyze_sentiment(df):
    """Rule-based sentiment analysis and summary results."""
    neg_words = {'bad','worst','poor','terrible','horrible','waste','hate','disappointed','cheap'}
    pos_words = {'good','great','excellent','amazing','love','perfect','nice','beautiful','best'}

    sentiments = []
    sentiment_scores = []  # +1 Positive, 0 Neutral, -1 Negative

    for text in df['preprocessed_review']:
        words = set(text.split())
        if words & neg_words:
            sentiments.append('Negative')
            sentiment_scores.append(-1)
        elif words & pos_words:
            sentiments.append('Positive')
            sentiment_scores.append(1)
        else:
            sentiments.append('Neutral')
            sentiment_scores.append(0)

    df['sentiment'] = sentiments
    df['sentiment_score'] = sentiment_scores

    counts = df['sentiment'].value_counts().to_dict()
    total = len(df)
    pos = counts.get('Positive', 0)
    neg = counts.get('Negative', 0)
    neu = counts.get('Neutral', 0)
    pos_percent = round((pos / total) * 100, 2) if total else 0

    # Recommendation logic
    if neg / total > 0.25:
        recommendation = "Not Recommended"
    elif pos_percent > 75:
        recommendation = "Highly Recommended"
    else:
        recommendation = "Needs Improvement"

    results = {
        "total_reviews": total,
        "sentiment_counts": {"Positive": pos, "Negative": neg, "Neutral": neu},
        "positive_percent": pos_percent,
        "recommendation": recommendation
    }

    return results, df

<script>
document.addEventListener('DOMContentLoaded', function() {
    const bridge = document.getElementById('chartData');
    const pos = parseInt(bridge.getAttribute('data-pos'));
    const neg = parseInt(bridge.getAttribute('data-neg'));
    const neu = parseInt(bridge.getAttribute('data-neu'));
    const reviews = JSON.parse(bridge.getAttribute('data-json'));
    const totalReviews = pos + neg + neu;

    let currentSentimentFilter = 'All';
    const baseColors = ['#2ecc71', '#e74c3c', '#f1c40f'];
    const dimColor = '#d3d3d3';

    // PIE CHART
    const pieChart = new Chart(document.getElementById('pieChart'), {
        type: 'pie',
        data: {
            labels: ['Positive','Negative','Neutral'],
            datasets:[{ data:[pos,neg,neu], backgroundColor:[...baseColors], hoverOffset:10 }]
        },
        options: {
            responsive:true,
            maintainAspectRatio:false,
            plugins:{
                legend:{ position:'bottom' },
                tooltip:{
                    callbacks:{
                        label: function(ctx){
                            const c = ctx.dataset.data[ctx.dataIndex];
                            const label = ctx.label;
                            const percent = Math.round((c/totalReviews)*100);
                            return `${label}: ${c} review${c!=1?'s':''} (${percent}%)`;
                        }
                    }
                }
            },
            onClick:(evt,elements)=>{
                if(elements.length>0){
                    const idx=elements[0].index;
                    const sentiments=['Positive','Negative','Neutral'];
                    currentSentimentFilter = currentSentimentFilter===sentiments[idx] ? 'All' : sentiments[idx];
                    updatePieColors();
                    applyFilters();
                }
            }
        }
    });

    function updatePieColors(){
        const colors=['Positive','Negative','Neutral'].map((s,i)=> currentSentimentFilter==='All'||currentSentimentFilter===s ? baseColors[i] : dimColor);
        pieChart.data.datasets[0].backgroundColor = colors;
        pieChart.update();
    }

    // TREND CHART
    const trendValues = reviews.map(r => r.sentiment==='Positive'?1:(r.sentiment==='Negative'?-1:0));
    const trendColors = reviews.map(r => r.sentiment==='Positive'? '#2ecc71':(r.sentiment==='Negative'?'#e74c3c':'#f1c40f'));
    const labels = reviews.map((_,i)=>`Review ${i+1}`);

    new Chart(document.getElementById('trendChart'),{
        type:'line',
        data:{ labels, datasets:[{ label:'Sentiment Flow', data:trendValues, borderColor:'#6c5ce7', tension:0.3, fill:false, pointBackgroundColor:trendColors, pointRadius:6 }] },
        options:{ responsive:true, maintainAspectRatio:false, scales:{ y:{ min:-1,max:1,ticks:{ stepSize:1, callback:v=>v===1?'Positive':v===0?'Neutral':'Negative' } } } }
    });

    // FILTER FUNCTION
    function applyFilters(){
        const rows=document.querySelectorAll('.review-row');
        rows.forEach(row=>{
            const text=row.querySelector('.review-text').innerText.toLowerCase();
            const sentiment=row.querySelector('.badge').innerText;
            const matchesSentiment = currentSentimentFilter==='All'||sentiment===currentSentimentFilter;
            row.style.display=matchesSentiment?'':'none';
        });
    }

    document.getElementById('showAllBtn').addEventListener('click', function(){
        currentSentimentFilter='All';
        updatePieColors();
        applyFilters();
    });
});
</script>
