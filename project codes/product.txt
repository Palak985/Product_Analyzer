def download_nltk_resources():
    resources = ['punkt', 'stopwords', 'wordnet', 'omw-1.4']
    for res in resources:
        try:
            nltk.data.find(res)
        except LookupError:
            nltk.download(res)

# --- 4. PREPROCESSING & SENTIMENT LOGIC ---
def preprocess_text(text):
    if not text or pd.isna(text): return ""
    text = str(text).lower()
    # Remove punctuation and special characters
    text = re.sub(r'[^a-zA-Z\s]', '', text)
    return text.strip()

def analyze_sentiment(df):
    # Dictionary-based triggers
    neg_triggers = {'not', 'no', 'bad', 'worst', 'terrible', 'horrible', 'poor', 'disappointed', 'waste', 'money', 'hate', 'itchy', 'cheap'}
    pos_triggers = {'good', 'great', 'best', 'excellent', 'amazing', 'love', 'perfect', 'nice', 'beautiful', 'fabulous', 'pleasant'}

    results_list = [] 
    for text in df['preprocessed_review']:
        words = set(str(text).split())
        
        # Priority Logic: Check for Negative triggers first
        if any(w in neg_triggers for w in words):
            results_list.append('Negative')
        elif any(w in pos_triggers for w in words):
            results_list.append('Positive')
        else:
            results_list.append('Neutral')

    df['sentiment'] = results_list
    counts = df['sentiment'].value_counts().to_dict()
    total = len(df)
    
    # Calculate percentages
    pos_count = counts.get('Positive', 0)
    neg_count = counts.get('Negative', 0)
    neu_count = counts.get('Neutral', 0)
    
    pos_perc = (pos_count / total * 100) if total > 0 else 0
    neg_perc = (neg_count / total * 100) if total > 0 else 0

    # Business Advice Logic
    if neg_perc > 25:
        advice, color = "⚠️ CRITICAL: High customer dissatisfaction. Investigate quality immediately.", "#e74c3c"
    elif pos_perc > 75:
        advice, color = "⭐ SUCCESS: Outstanding feedback! Perfect for marketing.", "#2ecc71"
    elif neu_count > (total * 0.5):
        advice, color = "ℹ️ NOTICE: High neutral volume. Product needs more 'wow' factor.", "#f1c40f"
    else:
        advice, color = "✅ STABLE: Consistent performance. Maintain current operations.", "#3498db"

    analysis_results = {
        'total_reviews': total,
        'sentiment_counts': {'Positive': pos_count, 'Negative': neg_count, 'Neutral': neu_count},
        'positive_percent': round(pos_perc, 2),
        'smart_advice': advice,
        'advice_color': color
    }
    
    return analysis_results, df

@app.route('/')

@app.route('/register', methods=['GET', 'POST'])
def register():
    global USER_ID_COUNTER
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        if username in USERS:
            flash('Username exists!', 'error')
            return redirect(url_for('register'))
        USERS[username] = User(USER_ID_COUNTER, username, generate_password_hash(password))
        USER_ID_COUNTER += 1
        return redirect(url_for('login'))
    return render_template('register.html')

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        user = USERS.get(username)
        if user and check_password_hash(user.password, password): 
            login_user(user) 
            return redirect(url_for('dashboard'))
        flash('Invalid Login', 'error') 
    return render_template('login.html')

@app.route('/logout')
@login_required
def logout():
    logout_user()
    return redirect(url_for('login'))

@app.route('/dashboard')
@login_required 
def dashboard():
    return render_template('dashboard.html', username=current_user.username)

@app.route('/upload_file', methods=['POST'])
@login_required
def upload_file():
    try:
        file = request.files.get('csv_file')
        if not file or file.filename == '':
            flash('No file selected!')
            return redirect(url_for('dashboard'))

        # Read CSV and clean column headers (removes hidden spaces)
        df = pd.read_csv(file)
        df.columns = df.columns.str.strip()

        # SMART COLUMN DETECTION
        possible_cols = ['Review_Text', 'reviewDescription', 'Reviews', 'text', 'Review', 'summary']
        found_col = next((c for c in possible_cols if c in df.columns), None)

        if found_col:
            df = df.rename(columns={found_col: 'Review_Text'})
        else:
            flash(f"Error: Review column not found. Columns: {list(df.columns)}")
            return redirect(url_for('dashboard'))

        # DATA CLEANING & PREPROCESSING
        df['Review_Text'] = df['Review_Text'].fillna("No content") 
        df['preprocessed_review'] = df['Review_Text'].apply(preprocess_text)
        
        # ANALYSIS
        results, df_final = analyze_sentiment(df)
        
        # SAVE TO SESSION (Professional Download Method)
        session['report_data'] = df_final.to_dict(orient='records')

        return render_template('analysis_report.html', 
                               results=results, 
                               review_data=df_final.head(20).to_dict(orient='records'),
                               filename=file.filename)

    except Exception as e:
        flash(f'System Error: {str(e)}')
        return redirect(url_for('dashboard'))

@app.route('/download_report', methods=['POST'])
@login_required
def download_report():
    data = session.get('report_data')
    if not data: return redirect(url_for('dashboard'))
    
    df = pd.DataFrame(data)
    response = make_response(df.to_csv(index=False))
    response.headers["Content-Disposition"] = "attachment; filename=Report.csv"
    response.headers["Content-type"] = "text/csv"
    return response

if __name__ == '__main__':
    app.run(debug=True)

    def ml_sentiment(text):
        score = vader.polarity_scores(text)['compound']
        if score >= 0.05:
            return 'Positive'
        elif score <= -0.05:
            return 'Negative'
        else:

    def analyze_sentiment(df):
        neg_words = {'bad','worst','poor','terrible','horrible','waste','hate','disappointed','cheap'}
        pos_words = {'good','great','excellent','amazing','love','perfect','nice','beautiful','best'}

        sentiments = []
        for text in df['preprocessed_review']:
            words = set(text.split())
            if words & neg_words:
                sentiments.append('Negative')
            elif words & pos_words:
                sentiments.append('Positive')
            else:
                sentiments.append('Neutral')

        df['sentiment'] = sentiments

        counts = df['sentiment'].value_counts().to_dict()
        total = len(df)

        pos = counts.get('Positive', 0)
        neg = counts.get('Negative', 0)
        neu = counts.get('Neutral', 0)

        pos_percent = round((pos / total) * 100, 2) if total else 0

        if neg / total > 0.25:
            advice = "⚠️ CRITICAL: High customer dissatisfaction detected."
            color = "#e74c3c"
            recommendation = "Not Recommended"
        elif pos_percent > 75:
            advice = "⭐ SUCCESS: Excellent customer satisfaction."
            color = "#2ecc71"
            recommendation = "Highly Recommended"
        else:
            advice = "ℹ️ STABLE: Mixed feedback. Improvements advised."
            color = "#f1c40f"
            recommendation = "Needs Improvement"

        results = {
            "total_reviews": total,
            "sentiment_counts": {"Positive": pos, "Negative": neg, "Neutral": neu},
            "positive_percent": pos_percent,
            "smart_advice": advice,
            "advice_color": color,
            "recommendation": recommendation
        }

        return results, df
