def download_nltk_resources():

        try:
            nltk.data.find(res)

            print(f"Downloading missing NLTK resource: {res}")

# --- 4. DATA PREPROCESSING (Refined for Accuracy) ---
try:
    STOP_WORDS = set(stopwords.words('english'))
    # Keeping negation words prevents "not good" from being read as "good"
    NEGATION_WORDS = {'not', 'no', 'never', 'neither', 'nor', 'none', 'but', 'only'}
    FILTERED_STOPWORDS = STOP_WORDS - NEGATION_WORDS
except LookupError:
    FILTERED_STOPWORDS = set()

PUNCTUATION = string.punctuation

def preprocess_text(text):
    if pd.isna(text) or text is None:
        return ""
    text = str(text).lower()
    text = re.sub(r'<.*?>', '', text) 
    text = re.sub(r'https?://\S+|www\.\S+', '', text) 
    text = text.translate(str.maketrans('', '', PUNCTUATION)) 
    words = word_tokenize(text)
    words = [word for word in words if word not in FILTERED_STOPWORDS and len(word) > 1]
    return " ".join(words)

@app.route('/register', methods=['GET', 'POST'])
def register():
    global USER_ID_COUNTER
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        if not username or not password:
            flash('Username and password are required.', 'error')
            return redirect(url_for('register'))
        if username in USERS:
            flash('Username already exists.', 'error')
            return redirect(url_for('register'))
        hashed_password = generate_password_hash(password)
        new_user = User(USER_ID_COUNTER, username, hashed_password)
        USERS[username] = new_user
        USER_ID_COUNTER += 1
        return redirect(url_for('login'))
    return render_template('register.html')

@app.route('/login', methods=['GET', 'POST'])
def login():
    if current_user.is_authenticated:
        return redirect(url_for('dashboard'))
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        user = USERS.get(username)
        if user and check_password_hash(user.password, password): 
            login_user(user) 
            return redirect(url_for('dashboard'))
        flash('Invalid username or password', 'error') 
    return render_template('login.html')

def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() == 'csv'

@app.route('/upload_file', methods=['POST'])
@login_required
def upload_file():
    session.pop('analysis_data', None)
    if 'csv_file' not in request.files:
        return redirect(url_for('dashboard'))

    file = request.files['csv_file']
    if file.filename == '' or not allowed_file(file.filename):
        return redirect(url_for('dashboard'))

    try:
        df = pd.read_csv(file)
        REVIEW_COLUMN = 'Review_Text' 
        
        if REVIEW_COLUMN not in df.columns:
            flash(f"Error: CSV must contain a column named '{REVIEW_COLUMN}'.", 'error')
            return redirect(url_for('dashboard'))
        
        df['preprocessed_review'] = df[REVIEW_COLUMN].apply(preprocess_text)
        results, df_analyzed = analyze_sentiment(df)
        
        # Prepare table and JSON for download
        review_list = df_analyzed[['Review_Text', 'preprocessed_review', 'sentiment']].head(15).to_dict('records')
        full_data_json = df_analyzed.to_json()
        
        response = make_response(render_template('analysis_report.html', 
                                               results=results, 
                                               username=current_user.username,
                                               filename=file.filename,
                                               review_data=review_list,
                                               full_data_json=full_data_json))
        
        response.headers['Cache-Control'] = 'no-cache, no-store, must-revalidate'
        return render_template('analysis_report.html', 
                       results=results, 
                       review_data=review_list, # This name must match the loop in HTML
                       full_data_json=df_analyzed.to_json())

    except Exception as e:
        flash(f'An error occurred: {e}', 'error')
        return redirect(url_for('dashboard'))
